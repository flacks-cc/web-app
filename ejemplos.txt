CREATE TABLE Usuarios (
    id_usuario SERIAL PRIMARY KEY,
    nombre_usuario VARCHAR(50) UNIQUE NOT NULL,
    nombre VARCHAR(50) NOT NULL,
    apellido_paterno VARCHAR(50) NOT NULL,
    apellido_materno VARCHAR(50),
    correo_electronico VARCHAR(100) UNIQUE NOT NULL,
    contraseña VARCHAR(255) NOT NULL,
    numero_telefono VARCHAR(10) UNIQUE NOT NULL
);

CREATE TABLE Roles (
    id_rol SERIAL PRIMARY KEY,
    nombre VARCHAR(50) UNIQUE NOT NULL
);

CREATE TABLE Usuarios_Roles (
    id_usuario_rol SERIAL PRIMARY KEY,
    id_usuario INT REFERENCES Usuarios(id_usuario) NOT NULL,
    id_rol INT REFERENCES Roles(id_rol) NOT NULL
);

CREATE TABLE Productos (
    id_producto SERIAL PRIMARY KEY,
    nombre VARCHAR(100) UNIQUE NOT NULL,
    descripcion VARCHAR(255),
    precio DECIMAL NOT NULL,
    stock INT NOT NULL
);

CREATE TABLE Servicios (
    id_servicio SERIAL PRIMARY KEY,
    nombre VARCHAR(100) UNIQUE NOT NULL,
    descripcion VARCHAR(255),
    precio DECIMAL NOT NULL,
    duracion INTERVAL
);

CREATE TABLE Reservaciones (
    id_reservacion SERIAL PRIMARY KEY,
    fecha DATE NOT NULL,
    hora TIME NOT NULL,
    activa BOOLEAN NOT NULL DEFAULT true,
    id_servicio INT REFERENCES Servicios(id_servicio),
    id_cliente INT REFERENCES Usuarios(id_usuario),
    id_empleado INT REFERENCES Usuarios(id_usuario)
);

CREATE TABLE Detalles_Productos (
    id_detalle_producto SERIAL PRIMARY KEY,
    cantidad INT NOT NULL,
    total DECIMAL NOT NULL,
    id_reservacion INT REFERENCES Reservaciones(id_reservacion) NOT NULL,
    id_producto INT REFERENCES Productos(id_producto) NOT NULL
);

CREATE TABLE Metodos_Pago (
    id_metodo_pago SERIAL PRIMARY KEY,
    nombre VARCHAR(50) UNIQUE NOT NULL
);

CREATE TABLE Tickets (
    id_ticket SERIAL PRIMARY KEY,
    fecha_hora_expedicion TIMESTAMP NOT NULL,
    monto_total DECIMAL NOT NULL,
    fecha_hora_pago TIMESTAMP,
    monto_pagado DECIMAL,
    cambio DECIMAL,
    id_metodo_pago INT REFERENCES Metodos_Pago(id_metodo_pago),
    id_empleado INT REFERENCES Usuarios(id_usuario),
    id_reservacion INT REFERENCES Reservaciones(id_reservacion) NOT NULL
);

CREATE TABLE Resenas (
    id_resena SERIAL PRIMARY KEY,
    mensaje VARCHAR(255),
    valoracion INT NOT NULL,
    fecha_hora TIMESTAMP NOT NULL,
    id_cliente INT REFERENCES Usuarios(id_usuario) NOT NULL,
    id_producto INT REFERENCES Productos(id_producto),
    id_servicio INT REFERENCES Servicios(id_servicio)
);

CREATE TABLE Mensajes_Contacto (
    id_mensaje SERIAL PRIMARY KEY,
    asunto VARCHAR(255) NOT NULL,
    mensaje VARCHAR(255) NOT NULL,
    adjunto VARCHAR(255),
    fecha_hora TIMESTAMP NOT NULL,
    id_cliente INT REFERENCES Usuarios(id_usuario) NOT NULL
);

CREATE TABLE Tarjetas (
    id_tarjeta SERIAL PRIMARY KEY,
    uid VARCHAR(8) UNIQUE NOT NULL,
    id_cliente INT REFERENCES Usuarios(id_usuario) UNIQUE NOT NULL
);
